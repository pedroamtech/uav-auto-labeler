# Self-contained. From the project root, in a fresh env:
#
#   conda create -n uav-auto-labeler python=3.11 -y
#   conda activate uav-auto-labeler
#   pip install -r requirements.txt
#
# No extra repos to clone, no `git` needed, works offline. See README.

# --- PyTorch: CUDA 12.8 wheels (RTX 50-series / Blackwell, sm_120) ------------
# The extra index serves the +cu128 wheels; PyPI serves everything else.
# CPU-only or another CUDA build: drop the --extra-index-url line and the
# `+cu128` suffixes (e.g. `torch==2.11.0`), or swap `cu128` for your toolkit.
--extra-index-url https://download.pytorch.org/whl/cu128
torch==2.11.0+cu128
torchvision==0.26.0+cu128

# --- YOLOv12 "ultralytics" fork (v8.3.63) ------------------------------------
# REQUIRED to load weights/best.pt (trained with this fork; its AAttn block
# uses a fused `qkv` layer the PyPI `ultralytics` package lacks). Not on PyPI,
# so it is vendored here as a pure-Python wheel. pip still resolves its normal
# runtime deps (numpy, opencv-python, pillow, pyyaml, scipy, pandas, tqdm, ...).
./vendor/ultralytics-8.3.63-py3-none-any.whl

# --- RF-DETR (only for --model rfdetr / union) ------------------------------
# On PyPI. Pulls transformers, supervision, av. Needs network to install
# (no offline wheel) and downloads its checkpoint (~386 MB) to
# ~/.roboflow/models/ on first use. Coexists with the YOLOv12 fork above in
# one env (tested: torch 2.11+cu128, numpy 2.x, transformers 5.x).
rfdetr
opencv-python

# --- This project ---------------------------------------------------------
tqdm
